  ```ts
  function mergeWithFallback(base: Dict, over: Dict, prefix: string, fallbacks: string[]): Dict {
    const out: Dict = {};
    for (const key of Object.keys(base)) {
      const b = base[key];
      const o = over[key];
      const path = prefix ? `${prefix}.${key}` : key;
      if (isDict(b)) {
        out[key] = mergeWithFallback(b, isDict(o) ? o : {}, path, fallbacks);
      } else if (Array.isArray(b)) {
        // массив — берём перевод целиком, только если это непустой массив той же длины строк;
        // иначе это не перевод, а недостающий ключ — используем base и отмечаем как fallback
        const validArray = Array.isArray(o) && o.length === b.length && o.every((x) => typeof x === 'string' && x.trim() !== '');
        out[key] = validArray ? o : b;
        if (!validArray) fallbacks.push(path);
      } else if (typeof o === 'string' && o.trim() !== '') {
        out[key] = o;
      } else {
        out[key] = b;
        fallbacks.push(path);
      }
    }
    for (const key of Object.keys(over)) {
      if (!(key in out)) out[key] = over[key];
    }
    return out;
  }
  ```
